def layout(p):
    return p.evaluate('''() => ({innerWidth,clientWidth:document.documentElement.clientWidth,
      scrollWidth:document.documentElement.scrollWidth,visualWidth:visualViewport.width,
      viewportMeta:document.querySelector('meta[name=viewport]')?.content||null,
      overflowing:[...document.querySelectorAll('input,select,textarea,button,a,table')].filter(e=>{
       const r=e.getBoundingClientRect();return r.width&&((r.right>document.documentElement.clientWidth+1)||r.left<0)
      }).map(e=>({tag:e.tagName,id:e.id,text:(e.textContent||'').slice(0,70),right:e.getBoundingClientRect().right}))})''')

def fits(u,result):
    return abs(result['innerWidth']-u.width)<=1 and result['scrollWidth']<=result['clientWidth']+1

def label_audit(p):
    return p.evaluate('''() => [...document.querySelectorAll('input:not([type=hidden]),select,textarea')]
      .filter(e=>e.getBoundingClientRect().width).map(e=>({id:e.id,type:e.type,
      label:e.labels?.length>0||!!e.getAttribute('aria-label')||!!e.getAttribute('aria-labelledby'),
      invalid:e.getAttribute('aria-invalid'),describedBy:e.getAttribute('aria-describedby'),
      missingDescriptions:(e.getAttribute('aria-describedby')||'').split(/\\s+/).filter(x=>x&&!document.getElementById(x))}))''')

@pytest.mark.parametrize('clipboard_mode',['native','rejected','unavailable'])
def test_acknowledgement_copy_and_fallback(ui,clipboard_mode):
    u=ui;p=u.page
    if clipboard_mode=='native':
        u.context.grant_permissions(['clipboard-read','clipboard-write'],origin=u.site['origin'])
    else:
        body="{writeText:async()=>{throw new DOMException('Synthetic denied','NotAllowedError')}}" if clipboard_mode=='rejected' else 'undefined'
        u.context.add_init_script("Object.defineProperty(navigator,'clipboard',{configurable:true,value:"+body+"});")
    accepted(u);identifier=p.locator('#complaint-id').inner_text().strip()
    p.get_by_role('button',name='Copy Complaint ID',exact=True).click()
    status=p.locator('#copy-complaint-id-status')
    if clipboard_mode=='native':
        expect(status).to_have_text('Complaint ID copied.')
        assert p.evaluate('navigator.clipboard.readText()')==identifier
    else:
        expect(status).to_contain_text('copy it manually')

    u.record('copy',{'mode':clipboard_mode,'status':status.inner_text(),'layout':layout(p),'text_status':p.locator('dl').inner_text()})
